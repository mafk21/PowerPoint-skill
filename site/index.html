python tools/wizard.py --goal pitch --topic "Loopwise" --claim "Turn cafe waste into packaging"

python tools/qa_deck.py deck.pptx                  # must print GREEN
python tools/social.py spec.json --handle @you     # Reel + Carousel pack
python database/query.py recommend pitch           # structures from the DB
